{{- /* Maths rendered to MathML at build time (KaTeX inside Hugo): no client JS, no maths CSS or fonts to download. */ -}} {{- $opts := dict "output" "mathml" "displayMode" (eq .Type "block") "throwOnError" false -}} {{- with try (transform.ToMath .Inner $opts) -}} {{- with .Err -}} {{- warnf "%s: maths error: %s" $.Page.File.Path . -}} {{ $.Inner }} {{- else -}} {{- $m := string .Value -}} {{- /* MathML Core ignores mathvariant other than "normal": map \mathbb, \mathbf, \mathfrak, \mathcal letters to their Unicode maths characters. */ -}} {{- if strings.Contains $m `mathvariant="` -}} {{- range $variant, $letters := hugo.Data.mathvariants -}} {{- if strings.Contains $m (printf `mathvariant="%s"` $variant) -}} {{- range $l, $g := $letters -}} {{- $m = replace $m (printf `%s` $variant $l) (printf `%s` $g) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* \lvert, \rvert and | as delimiters: fence spacing, not the "divides" relation spacing. */ -}} {{- $m = replace $m `∣` `∣` -}} {{- $m | safeHTML -}} {{- end -}} {{- end -}}
